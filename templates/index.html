async function sendMessage() { const input = document.getElementById("messageInput"); const message = input.value.trim(); if (!currentChat) { alert("Create a chat first"); return; } if (!message) return; const messages = document.getElementById("messages"); // User message const userMsg = document.createElement("div"); userMsg.classList.add("message", "user"); userMsg.innerText = message; messages.appendChild(userMsg); // Bot message container const botMsg = document.createElement("div"); botMsg.classList.add("message", "assistant"); // Loader const loader = document.createElement("div"); loader.classList.add("loader"); loader.innerHTML = ""; botMsg.appendChild(loader); messages.appendChild(botMsg); input.value = ""; messages.scrollTop = messages.scrollHeight; try { const res = await fetch("/send_message", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chat_id: currentChat, message: message }) }); if (!res.ok) { botMsg.innerText = "Error getting response"; return; } const data = await res.json(); botMsg.innerHTML = ""; botMsg.innerText = data.reply; } catch (error) { botMsg.innerHTML = ""; botMsg.innerText = "Something went wrong"; } messages.scrollTop = messages.scrollHeight; loadChats(); }
